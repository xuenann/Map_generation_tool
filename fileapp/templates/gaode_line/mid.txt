    function ContextMenu(map) {
        var me = this;
        //地图中添加鼠标工具MouseTool插件
        this.mouseTool = new AMap.MouseTool(map);
        this.contextMenuPositon = null;
        
        var content = [];
        content.push("<div class='info context_menu'>");
        content.push("  <p class='split_line' onclick='menu.distanceMeasureMenu()'>距离量测</p>");
        content.push("  <p onclick='menu.addMarkerMenu()'>添加标记</p>");
        content.push("</div>");

        //通过content自定义右键菜单内容
        this.contextMenu = new AMap.ContextMenu({isCustom: true, content: content.join('')});

        //地图绑定鼠标右击事件——弹出右键菜单
        map.on('rightclick', function (e) {
            if (isDrawing) {
                me.contextMenu.close();
                // 结束当前绘制但不关闭工具，允许继续绘制新线
                me.mouseTool.rule();
            } else {
                me.contextMenu.open(map, e.lnglat);
                me.contextMenuPositon = e.lnglat; //右键菜单位置
            }
        });
    }

    ContextMenu.prototype.distanceMeasureMenu = function () {  //右键菜单距离量测
        if (isDrawing) {
            // 如果正在绘制，先关闭当前绘制
            this.mouseTool.close();
        }
        // 开始新的绘制
        this.mouseTool.rule();
        this.contextMenu.close();
        // 标记当前正在绘制距离量测
        isDrawing = true;
    };

    // 监听键盘ESC键，退出绘制功能
    document.addEventListener('keydown', function(e) {
        if (e.key === 'Escape' || e.keyCode === 27) {
            // 按下ESC键，停止所有绘制
            menu.mouseTool.close();
            // 重置绘制状态
            isDrawing = false;
        }
    });


    //创建点标记右键菜单
    var markerContextMenu = new AMap.ContextMenu();
    markerContextMenu.addItem("删除点标记", function () {
        if (currentMarker) {
            // 从地图上移除点标记
            map.remove(currentMarker);
            // 从数组中移除点标记
            var index = allMarkers.indexOf(currentMarker);
            if (index > -1) {
                allMarkers.splice(index, 1);
            }
            // 清空当前标记引用
            currentMarker = null;
            // 关闭右键菜单
            markerContextMenu.close();
        }
    }, 0);
    ContextMenu.prototype.addMarkerMenu = function () {  //右键菜单添加Marker标记
        this.mouseTool.close();
        var marker = new AMap.Marker({
            map: map,
            position: this.contextMenuPositon || map.getCenter(), //基点位置
            icon: 'https://a.amap.com/jsapi_demos/static/demo-center/icons/poi-marker-red.png',
            offset: new AMap.Pixel(-9, -23) // 设置偏移，使标记中心点与点击位置对齐
        });
        // 将新创建的点标记添加到数组中
        allMarkers.push(marker);
        this.contextMenu.close();
        //绑定鼠标右击事件——弹出右键菜单
        marker.on('rightclick', function (e) {
            markerContextMenu.open(map, e.lnglat);
            // 将当前点标记存储到全局变量中，以便删除时使用
            currentMarker = this;
        });
    };
